set(PROJECT_SOURCES
    main.cpp
    MainWindow.cpp

    ../include/MainWindow.h

    ui/DisplayInterfaceWidget.cpp
    ../include/ui/DisplayInterfaceWidget.h

    ui/MatchResultWidget.cpp
    ../include/ui/MatchResultWidget.h

    ui/ClickStockWidget.cpp
    ../include/ui/ClickStockWidget.h

    data/DataBase.cpp
    ../include/data/DataBase.h

    data/DataPath.cpp
    ../include/data/DataPath.h

    core/KLineCalculate.cpp
    ../include/core/KLineCalculate.h

    ../include/model/KLineData.h
    ../include/model/KLineTime.h
)

add_executable(${PROJECT_NAME}
    ${PROJECT_SOURCES}
)

# 头文件目录
target_include_directories(${PROJECT_NAME}
    PRIVATE
        ${CMAKE_SOURCE_DIR}/include
)

# 链接 Qt Widgets
target_link_libraries(${PROJECT_NAME}
    PRIVATE
        Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Sql
)

# Windows 下生成 GUI 程序，不弹控制台
if(WIN32)
    set_target_properties(${PROJECT_NAME} PROPERTIES
        WIN32_EXECUTABLE TRUE
    )
endif()
